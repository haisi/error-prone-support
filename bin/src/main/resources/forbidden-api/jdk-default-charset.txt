# jdk-default-charset: JDK APIs that implicitly encode/decode using the JVM's platform default
# charset instead of an explicit java.nio.charset.Charset. The default charset varies by OS,
# locale and JVM flags (see JEP 400), so code relying on it is not portable.
#
# Verified against the JDK 25 API (java.base module), 2026-08-16, via `javap` against the
# installed JDK - each entry below has a corresponding Charset-accepting overload that should be
# used instead, except FileReader/FileWriter#<init>(FileDescriptor) which have none (flagged
# anyway, since they still depend on the default charset).

java.lang.String#<init>(byte[]) @ Platform default charset; use new String(bytes, StandardCharsets.XXX) or new String(bytes, someCharset)
java.lang.String#<init>(byte[], int, int) @ Platform default charset; use the (byte[], int, int, Charset) overload
java.lang.String#getBytes() @ Platform default charset; use getBytes(StandardCharsets.XXX) or getBytes(Charset)

java.io.InputStreamReader#<init>(java.io.InputStream) @ Platform default charset; use the (InputStream, Charset) overload
java.io.OutputStreamWriter#<init>(java.io.OutputStream) @ Platform default charset; use the (OutputStream, Charset) overload

java.io.FileReader#<init>(java.lang.String) @ Platform default charset; use the (String, Charset) overload
java.io.FileReader#<init>(java.io.File) @ Platform default charset; use the (File, Charset) overload
java.io.FileReader#<init>(java.io.FileDescriptor) @ Platform default charset; no Charset-accepting overload exists for FileDescriptor, prefer InputStreamReader with an explicit Charset

java.io.FileWriter#<init>(java.lang.String) @ Platform default charset; use the (String, Charset) overload
java.io.FileWriter#<init>(java.lang.String, boolean) @ Platform default charset; use the (String, Charset, boolean) overload
java.io.FileWriter#<init>(java.io.File) @ Platform default charset; use the (File, Charset) overload
java.io.FileWriter#<init>(java.io.File, boolean) @ Platform default charset; use the (File, Charset, boolean) overload
java.io.FileWriter#<init>(java.io.FileDescriptor) @ Platform default charset; no Charset-accepting overload exists for FileDescriptor, prefer OutputStreamWriter with an explicit Charset

java.util.Scanner#<init>(java.io.InputStream) @ Platform default charset; use the (InputStream, Charset) overload
java.util.Scanner#<init>(java.io.File) @ Platform default charset; use the (File, Charset) overload
java.util.Scanner#<init>(java.nio.file.Path) @ Platform default charset; use the (Path, Charset) overload
java.util.Scanner#<init>(java.nio.channels.ReadableByteChannel) @ Platform default charset; use the (ReadableByteChannel, Charset) overload
